```typescript
// 使用 SVG 绘制环形图
const QuotaRing = ({ used, limit }: { used: number; limit: number }) => {
    const percentage = limit > 0 ? (used / limit) * 100 : 0;
    const radius = 40;
    const circumference = 2 * Math.PI * radius;
    const offset = circumference - (percentage / 100) * circumference;

    return (
        <svg viewBox="0 0 100 100">
            <circle cx="50" cy="50" r={radius} fill="none" stroke="#e5e7eb" strokeWidth="8"/>
            <circle cx="50" cy="50" r={radius} fill="none"
                    stroke={percentage > 90 ? "#ef4444" : percentage > 70 ? "#f59e0b" : "#22c55e"}
                    strokeWidth="8" strokeDasharray={circumference} strokeDashoffset={offset}
                    transform="rotate(-90 50 50)"/>
            <text x="50" y="50" textAnchor="middle" dy=".3em" fontSize="14">
                {Math.round(percentage)}%
            </text>
        </svg>
    );
};
```

## 七、UpdateChecker

### 7.1 自动检测更新

UpdateChecker 使用 Tauri 的 `@tauri-apps/plugin-updater` 检测新版本：

```typescript
import { check, type Update } from "@tauri-apps/plugin-updater";
import { relaunch } from "@tauri-apps/plugin-process";

type Stage =
    | { kind: "idle" }
    | { kind: "checking" }
    | { kind: "latest" }          // 当前已是最新版
    | { kind: "available"; update: Update }
    | { kind: "downloading"; update: Update; percent: number }
    | { kind: "ready" }            // 下载完成，可以重启
    | { kind: "error"; message: string };

export function UpdateChecker({ onClose, onUpdateStarted }) {
    const [stage, setStage] = useState<Stage>({ kind: "idle" });

    const startCheck = async () => {
        setStage({ kind: "checking" });
        try {
            const update = await check();
            if (update) {
                setStage({ kind: "available", update });
            } else {
                setStage({ kind: "latest" });
            }
        } catch (e) {
            setStage({ kind: "error", message: String(e?.message ?? e) });
        }
    };

    const handleUpdate = async (update: Update) => {
        onUpdateStarted?.();
        setStage({ kind: "downloading", update, percent: 0 });
        await update.downloadAndInstall((event) => {
            if (event.event === "Progress") {
                setStage({ kind: "downloading", update, percent: event.data.chunkLength / total });
            }
        });
        setStage({ kind: "ready" });
        await relaunch();
    };
}
```

### 7.2 更新流程

```
用户点击"检查更新"
     │
     ▼
┌────────────┐
│  Checking   │ ← call check() API
└─────┬──────┘
      │
      ├─ 有更新 → ┌──────────────┐
      │           │  Available    │ ← 显示版本号和更新日志
      │           └─────┬────────┘
      │                 │ 用户点击"下载"
      │                 ▼
      │           ┌──────────────┐
      │           │  Downloading  │ ← 显示进度条
      │           └─────┬────────┘
      │                 │ 下载完成
      │                 ▼
      │           ┌──────────────┐
      │           │    Ready      │ ← 提示重启
      │           └─────┬────────┘
      │                 │
      │                 ▼
      │           relaunch() ← 自动重启应用
      │
      └─ 无更新 → ┌──────────────┐
                  │    Latest     │ ← 显示"已是最新版本"
                  └──────────────┘
```

```sql

```

- `response_choices`：记录模型返回的选项数量（如 `choices.length`）

### 8.2 006 — trace_id

```sql
ALTER TABLE request_logs ADD COLUMN trace_id TEXT;
```

**trace_id 的用途**：Agent 场景中一次用户请求可能触发多步 LLM 调用（RAG → 生成 → 补充搜索 → 再生成）。`trace_id` 通过 `Wali-Trace-Id` header 注入，将多步请求串联成一个完整链路。

## 十、面试问题

1. **DashboardPage 的 6 指标卡片数据来自 `statsApi.getDashboard()`，这个 API 的后端实现需要查询多少次数据库？如何优化？**

   > 提示：当前可能是 6 个独立 SQL 查询（today_requests, today_tokens, total_requests 等）。优化方案：合并为一个 SQL 用 `SELECT SUM(CASE WHEN date=today THEN 1 ELSE 0 END) as today_req, SUM(...) as today_tokens, ...` 一次查询返回所有指标。

2. **LogsPage 的 `mode` 字段（chat/messages/responses）如何区分三种协议的日志？如果用户通过 `/v1/chat/completions` 请求但渠道类型是 Anthropic，mode 是什么？**

   > 提示：`mode` 记录的是**入站协议**（用户用什么协议发送请求），不是**出站协议**（渠道用什么协议转发）。入站是 Chat Completions 就记录 "chat"，入站是 Messages 就记录 "messages"。内部转换后的转发不影响 mode 的值。

3. **UpdateChecker 使用 Tauri 的 `@tauri-apps/plugin-updater`，这个插件的更新检查机制是什么？如何确保更新包的安全性？**

   > 提示：Tauri updater 从配置的 `endpoints` URL 获取更新信息（版本号、下载链接、签名），下载后用 `pubkey` 验证签名确保包未被篡改。签名验证使用 Ed25519 公钥加密，只有持有私钥的开发者才能生成有效更新包。

4. **trace_id 在多步 Agent 场景中如何串联？如果 RAG 问答内部触发了多次 LLM 调用（检索+生成+补充），这些调用如何共享同一个 trace_id？**

   > 提示：MCP Client 在首次请求时生成 trace_id 并通过 `Wali-Trace-Id` header 传递。WaLiAPI 内部的 RAG 调用也应携带相同 trace_id（当前实现中 RAG 内部调用通过 proxy 层处理，需要将 trace_id 传递到 proxy 层）。

5. **ChannelsPage 的拖拽排序优先级和 Dispatcher 的渠道选择逻辑的关系是什么？用户调整了拖拽顺序后，如何影响请求分发？**

   > 提示：`Dispatcher::select_channels()` 按 priority 排序选择渠道。拖拽排序直接修改 channel.priority 字段，影响选择顺序。priority 低（数值小）的渠道优先被选中——拖到列表顶部的渠道 priority 最小。
